4: Aim
5: To design, implement, and integrate real-time bidirectional communication into the existing Vyralix full-stack platform using WebSockets (Socket.IO on Node.js/Express backend and socket.io-client on React frontend). The implementation enables multiple connected browser clients to broadcast and receive chat messages instantly without manual browser refreshing or short-polling.
6: Theory and Architectural Concepts
7: 1. HTTP vs. WebSockets: Traditional HTTP operates on a stateless request-response paradigm where the client must always initiate requests. Real-time data delivery via HTTP polling creates significant network overhead and latency. In contrast, WebSockets provide a persistent, full-duplex TCP communication channel, allowing both the server and client to send messages spontaneously with minimal framing overhead.
8: 2. Socket.IO Abstraction: Socket.IO is an event-based library built on top of WebSockets and HTTP long-polling fallback. It provides connection reliability, automatic reconnection, and an intuitive publish-subscribe event API using `socket.emit()` for emitting targeted events and `io.emit()` for broadcasting messages to all active client sockets.
9: 3. Shared Server Architecture: Instead of creating a separate server process for WebSockets, the Node.js `http` module wraps the Express application instance (`http.createServer(app)`). Socket.IO binds directly to this HTTP server, allowing both standard Express REST APIs and WebSocket handshakes to coexist on the exact same port (5000).
10: 4. React Lifecycle Integration: In the React client, `useRef` is leveraged to store the persistent socket instance across component renders. A `useEffect` hook handles the connection lifecycle upon mount and executes a cleanup function on unmount (`socket.disconnect()`), preventing memory leaks and duplicate socket connections.

const app = express();

// Express REST API routes (Exp 4-6)
app.use("/api/posts", postRoutes);

// Wrap Express with HTTP Server & Bind Socket.IO (Exp 8)
const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: "http://localhost:5173" }
});

// Real-Time Event Handlers
io.on("connection", (socket) => {
  console.log("A user connected:", socket.id);

  // Broadcast incoming chat message to all connected clients
  socket.on("chatMessage", (message) => {
    io.emit("chatMessage", message);
  });

  socket.on("disconnect", () => {
    console.log("A user disconnected");
  });
});

const PORT = process.env.PORT || 5000;
server.listen(PORT, () => {
  console.log(`Server running on port ${PORT}`);
});
14: 2. Frontend React Chat Component (vyralix-app/src/components/Chat.jsx)
15: import React, { useState, useEffect, useRef } from "react";
import { io } from "socket.io-client";
import "./Chat.css";

export default function Chat() {
  const [username, setUsername] = useState("");
  const [message, setMessage] = useState("");
  const [messages, setMessages] = useState([]);
  const [connected, setConnected] = useState(false);

  const socketRef = useRef(null);

  useEffect(() => {
    // Connect to Socket.IO server on port 5000
    const socket = io("http://localhost:5000");
    socketRef.current = socket;

    socket.on("connect", () => {
      setConnected(true);
    });

    socket.on("disconnect", () => {
      setConnected(false);
    });

    socket.on("chatMessage", (data) => {
      setMessages((prevMessages) => [...prevMessages, data]);
    });

    // Cleanup on unmount
    return () => {
      socket.disconnect();
    };
  }, []);

  const sendMessage = (event) => {
    event.preventDefault();
    if (!username.trim() || !message.trim()) return;

    if (socketRef.current) {
      socketRef.current.emit("chatMessage", {
        username: username.trim(),
        message: message.trim(),
      });
    }
    setMessage(""); // clear input field, retain username
  };

  return (
    <div className="chat-container">
      <h2>Vyralix Live Community Chat</h2>
      <p className="status">
        Status: {connected ? (
          <span className="status-connected">● Connected</span>
        ) : (
          <span className="status-connecting">○ Connecting...</span>
        )}
      </p>

      <div className="messages-box">
        {messages.length === 0 ? (
          <p className="no-messages">No messages yet. Start the conversation!</p>
        ) : (
          messages.map((msg, idx) => (
            <div key={idx} className="message-item">
              <strong className="message-user">{msg.username}:</strong>{" "}
              <span className="message-text">{msg.message}</span>
            </div>
          ))
        )}
      </div>

      <form onSubmit={sendMessage} className="chat-form">
        <input
          type="text"
          placeholder="Enter your name"
          value={username}
          onChange={(e) => setUsername(e.target.value)}
          className="chat-input"
        />
        <div className="message-input-row">
          <input
            type="text"
            placeholder="Type a message..."
            value={message}
            onChange={(e) => setMessage(e.target.value)}
            className="chat-input"
          />
          <button type="submit" className="chat-send-btn">Send</button>
        </div>
      </form>
    </div>
  );
}
